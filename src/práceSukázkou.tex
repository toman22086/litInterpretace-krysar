\section{Práce s~ukázkou}

\subsection{Zasazení ukázky do kontextu díla}
Ukázka pochází ze začátku knihy, představuje krysaře a~Agnes, a~uvádí čtenáře so do děje.

\subsection{Časoprostor}
\begin{itemize}
    \item \textbf{Čas:} Není napřímo zmíněný, pravděpodobně středověk.
	\item \textbf{Prostor:} město Hammeln a~jeho okolí
\end{itemize}

\subsection{Téma a~motivy}
\begin{itemize}
    \item \textbf{Téma}
    \begin{itemize}
        \item \textbf{Hlavní:} Krysařova zklamaná láska
        \item \textbf{Vedlejší:} Nedodržení slova a~následky
    \end{itemize}
    \item \textbf{Motivy:} krysy, píšťala, zrada, hamižnost, hora Koppel, země sedmihradské
\end{itemize}

\subsection{Charakteristika postav}
\begin{itemize}
	\item \textbf{Krysař} -- Kysař je popsán jako vysoký, štíhlý, tmavooký muž , s drobnýma, jemnýma rukama, oblečený v přiléhavém sametovém kabátci a úzkých nohavicích, který cestuje bezezbraně, pouze se svou zdobnou píšťalou. V knize se chová vážně, sebevědomě a samotářsky. Hluboce miluje Agnes.
	\item \textbf{Agnes} -- Agnes je popsána jako mladá, krázná dívka která má milence -- Dlouhého Kristiána, na toho však zapomíná v přítomnosti krysaře do kterého se zamilovala. Celou knihu se chová vesele, zvídavě a její smích je přírovnán ke zvonům vzkříšení, kromě konce kde se dozvídá, že s Kristiánem čeká dítě, stává se smutnou, nemluvnou a ze zoufalství páchá seberaždu.
\end{itemize}

\subsection{Kompoziční výstavba}
Kompozice je \textbf{chronologická} -- děj na sebe navazuje postupně.

\subsection{Vypravěč a~forma}
\textbf{Vypravěč} -- \textbf{Er forma}

\subsection{Typy promluv}
V~knize se vyskytyjí převážně \textbf{dialogy} a~občas \textbf{monology}, které jsou psané \textbf{přímou řečí}.

\subsection{Veršová výstavba}
 Nevyskytuje se, kniha je psána prózou

\subsection{Jazyková rovina}
Viktor Dyk napsal knihu \textbf{spisovnou češtinou}. Text je psán krátkými, jednoduchými větami. Obsahuje \textbf{knižní výrazy} např.: \textit{děvče, ponuré, lože}, \textbf{historismy} např.: \textit{konšelé, dukáty},  lze najít i \textbf{archaismy} např.: \textit{robili, hoře} a přechodníky.

\subsection{Umělecké prostředky, tropy a~figury}
\begin{itemize}
    \item \textbf{Gradace:} „Nejmenuji se; jsem nikdo. Jsem hůř než nikdo, jsem krysař.“
    \item \textbf{Přirovnání:} Jeho ruce byly \textit{drobné a~jemné jako ruce paní}.; Ženich \textit{byl bledý jako stěna}...
    \item \textbf{Inverze:} Odkud jsou, nevím.; Při svatbě Kateřiny \textit{objevila se pořádná krysa}.
	\item \textbf{Epizeuxis:} \textit{Hlodají} bez ustání; \textit{hlodají} dole ve sklepích, \textit{hlodají} tam, kde jich nevidíme.; „\textit{Píšťala},“ řekla pak pohrdavě. „Pěkná \textit{píšťala}. Ale nic více než \textit{píšťala}.“
	\item \textbf{Personifikace:} Krysař ukázal na \textit{píšťalu, která oživla} tímto pohybem.
\end{itemize}
