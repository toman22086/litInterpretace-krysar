\subsection*{2.~Osobnost a~tvorba autora, zasazení}
\begin{itemize}
    \item \textbf{O autorovi}\cite[150]{didaktis:literatura2}\cite{wiki:Viktor_Dyk}
    \begin{itemize}
        \item Vystudoval Právnickou fakultu Karlovy Univerzity
        \item Působil jako novinář, překladatel, spisovatel a~nacionalistický politik
        \item Od roku 1907 se podílel na časopisu Lumír
        \item Byl ovlivněn myšlenkami anarchismu
    \end{itemize}
    \item \textbf{Nejznámější literární díla:} Krysař, Zmoudření Dona Quijota, Devátá vlna
    \item \textbf{Zasazení autora do literárního období:} Přelom 19. a~20. století -- Moderna, patří k~autorům označovaným jako tzv.~generace buřičů

\end{itemize}
